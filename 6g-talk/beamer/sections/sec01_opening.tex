\begin{frame}{Six generations in forty years}
  \centering
  \includegraphics[width=0.78\textwidth]{fig_evolution_timeline.png}
  \begin{itemize}\itemsep1pt
    \item Each decade: a new generation, a $10$--$100\times$ peak-rate jump
    \item The differentiator each time was the \alert{physical-layer signal
          processing choice}: FDMA $\to$ TDMA $\to$ CDMA $\to$ OFDM/MIMO $\to$ mMIMO
  \end{itemize}
  \note{[S2, 1.5 min | clock 0:00:30] Walk the timeline left to right; stress
  that generations are remembered by their PHY: 3G is CDMA, 4G is OFDM+MIMO.
  Punchline: the 6G research window is open right now --- ITU framework 2023,
  3GPP study underway, spec by about 2029. The audience's papers written this
  year land exactly in that window.}
\end{frame}

\begin{frame}{Why signal processing is the bottleneck \emph{and} the enabler}
  \begin{itemize}\itemsep2pt
    \item At sub-6\,GHz we operate close to Shannon: coding, MIMO, scheduling
          are mature $\Rightarrow$ gains must come from \alert{new dimensions}
    \item New dimensions on offer: bandwidth (THz), aperture (ultra-massive
          arrays), \emph{environment} (RIS), \emph{function} (sensing),
          \emph{semantics} (AI)
    \item Hardware gets \emph{worse} as we go up in frequency --- DSP must
          compensate (impairment-aware design)
  \end{itemize}
  \centering\includegraphics[width=0.88\textwidth]{fig_roadmap.png}
  \note{[S5, 1 min | clock 0:05] The thesis slide. Every box on this roadmap is
  a DSP problem wearing a systems costume. Point at the D1--D5 dots: five live
  MATLAB experiments, one after each core section. Transition: ``New dimensions
  start with new spectrum.''}
\end{frame}
