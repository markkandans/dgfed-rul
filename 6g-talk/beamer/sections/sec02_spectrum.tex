% --- S7 ------------------------------------------------------------------
\begin{frame}{FR3 / upper mid-band: the pragmatic 6G band}
  \begin{itemize}\itemsep2pt
    \item 7--15\,GHz: $2$--$4\times$ more bandwidth than C-band with
          \emph{manageable} path loss --- the coverage--capacity compromise
    \item Same physical aperture $\Rightarrow$ $4$--$9\times$ more antenna
          elements than at 3.5\,GHz: massive MIMO scales \emph{naturally}
    \item Coexistence with incumbents (satellite, radar) $\Rightarrow$ sharing
          and sensing become part of the PHY job
    \item DSP price: wider bandwidth $\times$ bigger arrays $=$ beam squint,
          calibration, channel-estimation burden
  \end{itemize}
  \note{[S7, 1 min | clock 0:07] Explain why operators love FR3: reuse the
  macro grid. The antenna-count argument: element spacing scales with
  wavelength, so the same rooftop panel holds an order of magnitude more
  elements. Foreshadow Section 4 (arrays) via the last bullet.}
\end{frame}

      \item \textbf{Lesson 2:} beam training overhead is a \emph{rate} tax

% --- S9 ------------------------------------------------------------------
\begin{frame}{Sub-THz channels: 100--300\,GHz}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{equation*}
      \mathrm{PL}(f,d) =
      \underbrace{\left(\tfrac{4\pi f d}{c}\right)^{2}}_{\text{spreading}}
      \cdot
      \underbrace{e^{\kappa_{\mathrm{abs}}(f)\,d}}_{\text{molecular abs.}}
    \end{equation*}
    \begin{itemize}\itemsep1pt
      \item O$_2$/H$_2$O resonances at 60, 119, 183, 325\,GHz carve
            \alert{transmission windows}
      \item Channel: sparse, specular, LoS-dominated
      \item Consequences: short links, pencil beams mandatory
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=\textwidth]{fig_thz_absorption.png}
  \end{columns}
  \note{[S9, 1.5 min | clock 0:09] The one equation to remember: spreading loss
  grows with $f^2$ at fixed aperture, and the exponential absorption term is
  frequency-selective. Point at W/D/G windows on the plot --- those are where
  sub-THz systems will live. Sparsity is good news for compressed sensing
  later (Section 4).}
\end{frame}

% --- S10 -----------------------------------------------------------------
\begin{frame}{Hardware impairments dominate at high bands}
  \begin{columns}[T]
    \column{0.52\textwidth}
    Rapp PA model (AM/AM):
    \begin{equation*}
      g(A)=\frac{G\,A}{\left[1+\left(\frac{GA}{A_{\mathrm{sat}}}\right)^{2p}
            \right]^{1/2p}}
    \end{equation*}
    Wiener phase noise:
    \begin{equation*}
      \varphi[n{+}1]=\varphi[n]+w[n],\;\;
      w\sim\mathcal N\!\left(0,\,4\pi^{2}f_c^{2}c\,T_s\right)
    \end{equation*}
    \begin{itemize}\itemsep1pt
      \item PA efficiency falls with $f$; back-off $\Rightarrow$ PAPR is a
            first-class metric
      \item Phase noise power $\propto f_c^{2}$; low-resolution ADCs to save
            power
    \end{itemize}
    \column{0.46\textwidth}
    \centering
    \includegraphics[width=\textwidth]{fig_pa_rapp.png}
  \end{columns}
  \note{[S10, 1.5 min | clock 0:10:30] ``Dirty RF'' mindset: at 140 GHz the
  waveform must be designed FOR the hardware. Two knobs to remember: PAPR
  (decides PA back-off, hence coverage) and phase-noise robustness (decides
  subcarrier spacing / single-carrier choice). This is the bridge to
  waveforms.}
\end{frame}

\begin{frame}{Implications for waveform design}
  \centering
  \small
  \begin{tabular}{@{}lcccc@{}}
    \toprule
    \textbf{Requirement} & sub-6 & FR3 & mmWave & sub-THz \\
    \midrule
    Low PAPR                  & nice  & nice  & want & \alert{need} \\
    Phase-noise robustness    & ---   & ---   & want & \alert{need} \\
    Doppler robustness        & HST   & HST   & want & want \\
    Spectral confinement      & want  & want  & nice & nice \\
    MIMO compatibility        & \alert{need} & \alert{need} & \alert{need} & want \\
    \bottomrule
  \end{tabular}
  \vspace{2mm}
  \begin{itemize}
    \item No single waveform wins every column $\Rightarrow$ expect
          \alert{scenario-dependent waveform choices} in 6G
    \item CP-OFDM's weak spots are exactly the columns that harden at high bands
  \end{itemize}
  \note{[S11, 1 min | clock 0:12] Read the table column-wise: the further
  right, the more the requirements shift toward PAPR and phase noise, where
  OFDM is weakest. Transition: ``So let us give OFDM its fair trial ---
  Section 3.''}
\end{frame}
