\begin{frame}{The LEO signal processing problem: Doppler}
  \begin{equation*}
    f_D(t)=\frac{f_c}{c}\,v_{\mathrm{rel}}(t),
    \qquad v_{\mathrm{sat}}\approx 7.6\,\text{km/s}
    \;\Rightarrow\; \pm 25\,\text{ppm}
    \;\;(\pm 50\,\text{kHz at 2\,GHz})
  \end{equation*}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}\itemsep1pt
      \item Doppler \emph{rate} up to $\sim$kHz/s near zenith; RTT 2--25\,ms
            breaks HARQ timing
      \item The gift: \alert{ephemeris is known} $\Rightarrow$ pre-compensate,
            track only the residual
      \item DD-domain waveforms (Section 3) are natural fits here
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=\textwidth]{fig_leo_doppler.png}
  \end{columns}
  \note{[S57, 1 min | clock 1:16] The S-curve is computed from real orbital
  geometry (600 km). Read it: half the swing happens in the two minutes
  around zenith --- that is the tracking stress test. Then the twist that makes
  NTN tractable: unlike a car, a satellite tells you its velocity in advance;
  open-loop pre-compensation removes 95 percent of the problem.}
\end{frame}

% --- S59 -----------------------------------------------------------------
\begin{frame}{Optical wireless and VLC}
  \begin{itemize}\itemsep2pt
    \item IM/DD constraint: real, non-negative signals $\Rightarrow$
          DCO-/ACO-OFDM adaptations of familiar math
    \item LiFi: dense indoor capacity, security by physics (light stays in the
          room); FSO: fiber-class backhaul without the trench --- incl.\
          ground--satellite
    \item Challenges: alignment, ambient light, the uplink asymmetry
    \item Honest niche: industrial floors, cabins, RF-denied environments
  \end{itemize}
  \note{[S59, 1 min | clock 1:17:30] Position optical as a complement, not a
  competitor: where RF is denied or congested, photons commute. The DSP
  detail worth stating: IM/DD kills negative amplitudes, so OFDM needs DC
  bias or asymmetric clipping --- same toolbox, new constraint set.}
\end{frame}

\begin{frame}{Quantum-secured links (brief)}
  \begin{itemize}\itemsep2pt
    \item QKD: key distribution whose security rests on physics; fiber and
          satellite demos exist
    \item Realism: point-to-point, modest key rates $\Rightarrow$ practical
          path is \alert{hybrid} QKD + post-quantum cryptography
    \item For 6G: primarily a security-architecture topic, not a PHY-DSP one
          --- scoped to one slide on purpose
  \end{itemize}
  \note{[S60, 0.5 min | clock 1:18:30] Deliberately brief --- acknowledge the
  buzzword, place it correctly, move on. If asked in Q\&A: the DSP-adjacent
  part is CV-QKD receivers (coherent detection, error reconciliation), a
  legitimate niche for signal processors.}
\end{frame}
